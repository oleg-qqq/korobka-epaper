176:void Application::Run() {
177:    // Set the priority of the main task to 10
178:    vTaskPrioritySet(nullptr, 10);
179:
180:    const EventBits_t ALL_EVENTS =
181:        MAIN_EVENT_SCHEDULE | MAIN_EVENT_SEND_AUDIO | MAIN_EVENT_WAKE_WORD_DETECTED |
182:        MAIN_EVENT_VAD_CHANGE | MAIN_EVENT_CLOCK_TICK | MAIN_EVENT_ERROR |
183:        MAIN_EVENT_NETWORK_CONNECTED | MAIN_EVENT_NETWORK_DISCONNECTED | MAIN_EVENT_TOGGLE_CHAT |
184:        MAIN_EVENT_START_LISTENING | MAIN_EVENT_STOP_LISTENING | MAIN_EVENT_ACTIVATION_DONE |
185:        MAIN_EVENT_STATE_CHANGED | MAIN_EVENT_PLAYBACK_DRAINED;
186:
187:    while (true) {
188:        auto bits = xEventGroupWaitBits(event_group_, ALL_EVENTS, pdTRUE, pdFALSE, portMAX_DELAY);
189:
190:        if (bits & MAIN_EVENT_ERROR) {
191:            if (GetDeviceState() == kDeviceStateNotifying) {
192:                StopNotification();
193:            }
194:            SetDeviceState(kDeviceStateIdle);
195:            Alert(Lang::Strings::ERROR, last_error_message_.c_str(), "cancel",
196:                  Lang::Sounds::OGG_EXCLAMATION);
197:        }
198:
199:        if (bits & MAIN_EVENT_NETWORK_CONNECTED) {
200:            HandleNetworkConnectedEvent();
201:        }
202:
203:        if (bits & MAIN_EVENT_NETWORK_DISCONNECTED) {
204:            HandleNetworkDisconnectedEvent();
205:        }
206:
207:        if (bits & MAIN_EVENT_ACTIVATION_DONE) {
208:            HandleActivationDoneEvent();
209:        }
210:
211:        if (bits & MAIN_EVENT_STATE_CHANGED) {
212:            HandleStateChangedEvent();
213:        }
214:
215:        if (bits & MAIN_EVENT_PLAYBACK_DRAINED) {
216:            if (audio_service_.IsPlaybackIdle()) {
217:                notify_player_.OnPlaybackDrained();
218:            }
219:            // Deferred listening start (auto mode): the playback queue has
220:            // drained, so it is now safe to enable voice processing.
221:            if (pending_listening_start_ && GetDeviceState() == kDeviceStateListening &&
222:                audio_service_.IsPlaybackIdle()) {
223:                pending_listening_start_ = false;
224:                StartListeningAudio();
225:            }
226:        }
227:
228:        if (bits & MAIN_EVENT_TOGGLE_CHAT) {
229:            HandleToggleChatEvent();
230:        }
231:
232:        if (bits & MAIN_EVENT_START_LISTENING) {
233:            HandleStartListeningEvent();
234:        }
235:
236:        if (bits & MAIN_EVENT_STOP_LISTENING) {
237:            HandleStopListeningEvent();
238:        }
239:
240:        if (bits & MAIN_EVENT_SEND_AUDIO) {
241:            while (auto packet = audio_service_.PopPacketFromSendQueue()) {
242:                if (protocol_ && !protocol_->SendAudio(std::move(packet))) {
243:                    // Drop the remaining packets. Leaving them in the queue would
244:                    // stall the Opus codec task (it waits for queue space), which in
245:                    // turn deadlocks the whole audio input pipeline, as no new
246:                    // MAIN_EVENT_SEND_AUDIO event would ever be triggered again.
247:                    while (audio_service_.PopPacketFromSendQueue())
248:                        ;
249:                    break;
250:                }
251:            }
252:        }
253:
254:        if (bits & MAIN_EVENT_WAKE_WORD_DETECTED) {
255:            HandleWakeWordDetectedEvent();
256:        }
257:
258:        if (bits & MAIN_EVENT_VAD_CHANGE) {
259:            if (GetDeviceState() == kDeviceStateListening) {
260:                auto led = Board::GetInstance().GetLed();
261:                led->OnStateChanged();
262:            }
263:        }
264:
265:        if (bits & MAIN_EVENT_SCHEDULE) {
266:            std::unique_lock<std::mutex> lock(mutex_);
267:            auto tasks = std::move(main_tasks_);
268:            lock.unlock();
269:            for (auto& task : tasks) {
270:                task();
271:            }
272:        }
273:
274:        if (bits & MAIN_EVENT_CLOCK_TICK) {
275:            clock_ticks_++;
276:            auto display = Board::GetInstance().GetDisplay();
277:            display->UpdateStatusBar();
278:
279:            // Print debug info every 10 seconds
280:            if (clock_ticks_ % 10 == 0) {
281:                SystemInfo::PrintHeapStats();
282:                // SystemInfo::PrintTaskList();
283:                // SystemInfo::PrintTaskCpuUsage(pdMS_TO_TICKS(1000));
284:            }
285:        }
286:    }
287:}

355:    board.SetPowerSaveLevel(PowerSaveLevel::LOW_POWER);
356:

442:void Application::CheckNewVersion() {
443:    const int MAX_RETRY = 10;
444:    int retry_count = 0;
445:    int retry_delay = 10;  // Initial retry delay in seconds
446:
447:    auto& board = Board::GetInstance();
448:    while (true) {
449:        auto display = board.GetDisplay();
450:        display->SetStatus(Lang::Strings::CHECKING_NEW_VERSION);
451:
452:        auto check = ota_->CheckVersion();
453:        if (!check) {
454:            retry_count++;
455:            if (retry_count >= MAX_RETRY) {
456:                ESP_LOGE(TAG, "Too many retries, exit version check");
457:                return;
458:            }
459:
460:            const auto& err = check.error();
461:            char error_message[160];
462:            int error_message_length =
463:                snprintf(error_message, sizeof(error_message), "%s", err.ToString().c_str());
464:            if (error_message_length < 0 ||
465:                error_message_length >= static_cast<int>(sizeof(error_message))) {
466:                snprintf(error_message, sizeof(error_message), "%s", err.Message());
467:            }
468:
469:            char buffer[320];
470:            int alert_message_length =
471:                snprintf(buffer, sizeof(buffer), Lang::Strings::CHECK_NEW_VERSION_FAILED,
472:                         retry_delay, error_message);
473:            if (alert_message_length < 0 ||
474:                alert_message_length >= static_cast<int>(sizeof(buffer))) {
475:                snprintf(buffer, sizeof(buffer), "%s", err.Message());
476:            }
477:            Alert(Lang::Strings::ERROR, buffer, "cloud_off", Lang::Sounds::OGG_EXCLAMATION);
478:
479:            ESP_LOGW(TAG, "Check new version failed, retry in %d seconds (%d/%d)", retry_delay,
480:                     retry_count, MAX_RETRY);
481:            for (int i = 0; i < retry_delay; i++) {
482:                vTaskDelay(pdMS_TO_TICKS(1000));
483:                if (GetDeviceState() == kDeviceStateIdle) {
484:                    break;
485:                }
486:            }
487:            retry_delay *= 2;  // Double the retry delay
488:            continue;
489:        }
490:        retry_count = 0;
491:        retry_delay = 10;  // Reset retry delay
492:
493:        if (ota_->HasNewVersion()) {
494:            if (UpgradeFirmware(ota_->GetFirmwareUrl(), ota_->GetFirmwareVersion())) {
495:                return;  // This line will never be reached after reboot
496:            }
497:            // If upgrade failed, continue to normal operation
498:        }
499:
500:        // No new version, mark the current version as valid
501:        ota_->MarkCurrentVersionValid();
502:        if (!ota_->HasActivationCode() && !ota_->HasActivationChallenge()) {
503:            // Exit the loop if done checking new version
504:            break;
505:        }
506:
507:        display->SetStatus(Lang::Strings::ACTIVATION);
508:        // Activation code is shown to the user and waiting for the user to input
509:        if (ota_->HasActivationCode()) {
510:            ShowActivationCode(ota_->GetActivationCode(), ota_->GetActivationMessage());
511:        }
512:
513:        // This will block the loop until the activation is done or timeout
514:        for (int i = 0; i < 10; ++i) {
515:            ESP_LOGI(TAG, "Activating... %d/%d", i + 1, 10);
516:            esp_err_t err = ota_->Activate();
517:            if (err == ESP_OK) {
518:                break;
519:            } else if (err == ESP_ERR_TIMEOUT) {
520:                vTaskDelay(pdMS_TO_TICKS(3000));
521:            } else {
522:                vTaskDelay(pdMS_TO_TICKS(10000));
523:            }
524:            if (GetDeviceState() == kDeviceStateIdle) {
525:                break;
526:            }
527:        }
528:    }
529:}

531:void Application::InitializeProtocol() {
532:    auto& board = Board::GetInstance();
533:    auto display = board.GetDisplay();
534:    auto codec = board.GetAudioCodec();
535:
536:    display->SetStatus(Lang::Strings::LOADING_PROTOCOL);
537:
538:    if (ota_->HasMqttConfig()) {
539:        protocol_ = std::make_unique<MqttProtocol>();
540:    } else if (ota_->HasWebsocketConfig()) {
541:        protocol_ = std::make_unique<WebsocketProtocol>();
542:    } else {
543:        ESP_LOGW(TAG, "No protocol specified in the OTA config, using MQTT");
544:        protocol_ = std::make_unique<MqttProtocol>();
545:    }
546:
547:    protocol_->OnConnected([this]() { DismissAlert(); });
548:
549:    protocol_->OnNetworkError([this](const std::string& message) {
550:        last_error_message_ = message;
551:        xEventGroupSetBits(event_group_, MAIN_EVENT_ERROR);
552:    });
553:
554:    protocol_->OnIncomingAudio([this](std::unique_ptr<AudioStreamPacket> packet) {
555:        if (GetDeviceState() == kDeviceStateSpeaking) {
556:            audio_service_.PushPacketToDecodeQueue(std::move(packet));
557:        }
558:    });
559:
560:    protocol_->OnAudioChannelOpened([this, codec, &board]() {
561:        board.SetPowerSaveLevel(PowerSaveLevel::PERFORMANCE);
562:        if (protocol_->server_sample_rate() != codec->output_sample_rate()) {
563:            ESP_LOGW(TAG,
564:                     "Server sample rate %d does not match device output sample rate %d, "
565:                     "resampling may cause distortion",
566:                     protocol_->server_sample_rate(), codec->output_sample_rate());
567:        }
568:    });
569:
570:    protocol_->OnAudioChannelClosed([this, &board]() {
571:        board.SetPowerSaveLevel(PowerSaveLevel::LOW_POWER);
572:        Schedule([this]() {
573:            auto display = Board::GetInstance().GetDisplay();
574:            display->SetChatMessage("system", "");
575:            SetDeviceState(kDeviceStateIdle);
576:        });
577:    });
578:
579:    protocol_->OnIncomingJson([this, display](const cJSON* root) {
580:        // Parse JSON data
581:        auto type = cJSON_GetObjectItem(root, "type");
582:        if (!cJSON_IsString(type)) {
583:            ESP_LOGW(TAG, "Incoming JSON message has no type");
584:            return;
585:        }
586:        if (strcmp(type->valuestring, "notify") == 0) {
587:            auto audio_url = cJSON_GetObjectItem(root, "audio_url");
588:            if (!cJSON_IsString(audio_url) || audio_url->valuestring[0] == '\0') {
589:                ESP_LOGW(TAG, "Notify message requires audio_url");
590:                return;
591:            }
592:
593:            std::vector<NotifySubtitle> subtitles;
594:            auto subtitles_json = cJSON_GetObjectItem(root, "subtitles");
595:            if (subtitles_json != nullptr && !cJSON_IsArray(subtitles_json)) {
596:                ESP_LOGW(TAG, "Notify subtitles must be an array");
597:                return;
598:            }
599:            if (cJSON_IsArray(subtitles_json)) {
600:                cJSON* item = nullptr;
601:                cJSON_ArrayForEach (item, subtitles_json) {
602:                    auto start_ms = cJSON_GetObjectItem(item, "start_ms");
603:                    auto text = cJSON_GetObjectItem(item, "text");
604:                    if (!cJSON_IsNumber(start_ms) || start_ms->valuedouble < 0 ||
605:                        start_ms->valuedouble > std::numeric_limits<uint32_t>::max() ||
606:                        !cJSON_IsString(text)) {
607:                        ESP_LOGW(TAG, "Ignoring invalid notify subtitle");
608:                        continue;
609:                    }
610:                    subtitles.push_back({.start_ms = static_cast<uint32_t>(start_ms->valuedouble),
611:                                         .text = text->valuestring});
612:                }
613:            }
614:

897:void Application::HandleWakeWordDetectedEvent() {
898:    if (!protocol_) {
899:        return;
900:    }
901:
902:    auto state = GetDeviceState();
903:    auto wake_word = audio_service_.GetLastWakeWord();
904:    ESP_LOGI(TAG, "Wake word detected: %s (state: %d)", wake_word.c_str(), (int)state);

908:    } else if (state == kDeviceStateNotifying) {

922:            audio_service_.EnableWakeWordDetection(true);
923:        } else {
924:            // Play popup sound and start listening again
925:            play_popup_on_listening_ = true;
926:            SetListeningMode(GetDefaultListeningMode());
927:        }
928:    } else if (state == kDeviceStateActivating) {
929:        // Restart the activation check if the wake word is detected during activation
930:        SetDeviceState(kDeviceStateIdle);

934:void Application::BeginWakeWordInvoke(const std::string& wake_word) {
935:    // Must run in the main task with the device in idle state
936:    audio_service_.EncodeWakeWord();
937:
938:    // Always pass through the connecting state, even if the audio channel is
939:    // already opened. ContinueWakeWordInvoke() rejects any other state, so
940:    // skipping this transition would silently drop the wake word invocation.
941:    if (!SetDeviceState(kDeviceStateConnecting)) {
942:        // Wake word detection was stopped by the detection itself; restore it
943:        // so the device does not become unresponsive to wake words.
944:        audio_service_.EnableWakeWordDetection(true);
945:        return;
946:    }
947:
948:    if (!protocol_->IsAudioChannelOpened()) {
949:        // Schedule to let the state change be processed first (UI update),
950:        // then continue with OpenAudioChannel which may block for ~1 second
951:        Schedule([this, wake_word]() { ContinueWakeWordInvoke(wake_word); });
952:        return;
953:    }
954:    // Channel already opened, continue directly
955:    ContinueWakeWordInvoke(wake_word);

995:void Application::HandleStateChangedEvent() {
996:    DeviceState new_state = state_machine_.GetState();
997:    clock_ticks_ = 0;
998:    // Any state change invalidates a pending deferred listening start;
999:    // the Listening case below re-arms it when needed.
1000:    pending_listening_start_ = false;
1001:
1002:    auto& board = Board::GetInstance();
1003:    auto display = board.GetDisplay();
1004:    auto led = board.GetLed();
1005:    led->OnStateChanged();
1006:
1007:    switch (new_state) {
1008:        case kDeviceStateUnknown:
1009:        case kDeviceStateIdle:
1010:            // Keep a just-raised network error visible. SetDeviceState(idle)
1011:            // queues STATE_CHANGED after Alert(), and the idle handler would
1012:            // otherwise wipe the status, emotion, and chat message.
1013:            if (last_error_message_.empty()) {
1014:                display->SetStatus(Lang::Strings::STANDBY);
1015:                display->ClearChatMessages();  // Clear messages first
1016:                display->SetEmotion(
1017:                    "neutral");  // Then set emotion (wechat mode checks child count)
1018:            }
1019:            audio_service_.EnableVoiceProcessing(false);
1020:            audio_service_.EnableWakeWordDetection(true);
1021:            break;
1022:        case kDeviceStateConnecting:
1023:            display->SetStatus(Lang::Strings::CONNECTING);
1024:            display->SetEmotion("neutral");
1025:            display->SetChatMessage("system", "");
1026:            break;
1027:        case kDeviceStateListening:
1028:            display->SetStatus(Lang::Strings::LISTENING);
1029:            display->SetEmotion("neutral");
1030:
1031:            // Make sure the audio processor is running
1032:            if (play_popup_on_listening_ || !audio_service_.IsAudioProcessorRunning()) {
1033:                // For auto mode, wait for the playback queue to drain before enabling
1034:                // voice processing. This prevents audio truncation when STOP arrives
1035:                // late due to network jitter. Instead of blocking the main loop here,
1036:                // defer the start until MAIN_EVENT_PLAYBACK_DRAINED arrives.
1037:                if (listening_mode_ == kListeningModeAutoStop && !audio_service_.IsPlaybackIdle()) {
1038:                    pending_listening_start_ = true;
1039:                } else {
1040:                    StartListeningAudio();
1041:                }
1042:            } else {
1043:                ConfigureWakeWordForListening();
1044:            }
1045:            break;
1046:        case kDeviceStateSpeaking:
1047:            display->SetStatus(Lang::Strings::SPEAKING);
1048:
1049:            if (listening_mode_ != kListeningModeRealtime) {
1050:                audio_service_.EnableVoiceProcessing(false);
1051:                // Only AFE wake word can be detected in speaking mode
1052:                audio_service_.EnableWakeWordDetection(audio_service_.IsAfeWakeWord());
1053:            }
1054:            audio_service_.ResetDecoder();
1055:            break;
1056:        case kDeviceStateNotifying:
1057:            display->SetStatus(Lang::Strings::SPEAKING);
1058:            audio_service_.EnableVoiceProcessing(false);
1059:            audio_service_.EnableWakeWordDetection(audio_service_.IsAfeWakeWord());
1060:            break;
1061:        case kDeviceStateWifiConfiguring:
1062:            audio_service_.EnableVoiceProcessing(false);
1063:            audio_service_.EnableWakeWordDetection(false);
1064:            break;
1065:        default:
1066:            // Do nothing
1067:            break;
1068:    }
1069:}

1268:    if (!protocol_) {
1269:        return;
1270:    }
1271:
1272:    auto state = GetDeviceState();
1273:
1274:    if (state == kDeviceStateIdle) {
1275:        // May be called from outside the main task (e.g. board button
1276:        // callbacks), so schedule the invocation instead of running it here
1277:        Schedule([this, wake_word]() {
1278:            if (GetDeviceState() == kDeviceStateIdle) {
1279:                BeginWakeWordInvoke(wake_word);
1280:            }
1281:        });
1282:    } else if (state == kDeviceStateNotifying) {
1283:        Schedule([this, wake_word]() {
1284:            if (GetDeviceState() == kDeviceStateNotifying) {
1285:                StopNotification();
1286:                BeginWakeWordInvoke(wake_word);
1287:            }
1288:        });
1289:    } else if (state == kDeviceStateSpeaking) {
1290:        Schedule([this]() { AbortSpeaking(kAbortReasonNone); });
1291:    } else if (state == kDeviceStateListening) {
1292:        Schedule([this]() {
1293:            if (protocol_) {
1294:                protocol_->CloseAudioChannel();
1295:            }
1296:        });
1297:    }
